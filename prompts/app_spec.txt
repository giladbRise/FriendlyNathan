<project_specification>
  <project_name>RISE n8n Workflow AI Builder</project_name>

  <overview>
    A production-ready AI-powered web application that enables RISE employees to create n8n workflows through natural language descriptions. The system uses Google Gemini 3 Flash Preview to intelligently analyze available n8n nodes, generate optimal workflows, and create them directly in customer n8n instances via API. It provides comprehensive guidance for credentials, handles edge cases gracefully, and delivers seamless, working workflows ready for validation.
  </overview>

  <api_endpoints_summary>
    <authentication>
      - POST /api/auth/register - Register new user
      - POST /api/auth/login - Login and receive JWT
      - POST /api/auth/logout - Logout and invalidate token
      - POST /api/auth/refresh - Refresh JWT token
      - POST /api/auth/forgot-password - Request password reset
      - POST /api/auth/reset-password - Reset password with token
      - GET /api/auth/me - Get current user profile
      - PUT /api/auth/me - Update current user profile
      - PUT /api/auth/change-password - Change password
    </authentication>

    <n8n_instances>
      - GET /api/n8n-instances - List user's saved n8n instances
      - POST /api/n8n-instances - Save new n8n instance
      - PUT /api/n8n-instances/:id - Update n8n instance
      - DELETE /api/n8n-instances/:id - Delete n8n instance
      - POST /api/n8n-instances/validate - Validate n8n URL and API key
    </n8n_instances>

    <workflow_generation>
      - POST /api/workflows/generate - Generate and create workflow
      - GET /api/workflows/history - Get user's workflow history
      - GET /api/workflows/:id - Get specific workflow generation details
      - POST /api/workflows/:id/retry - Retry failed workflow generation
      - POST /api/nodes/discover - Discover available nodes from n8n instance
    </workflow_generation>

    <credential_guidance>
      - GET /api/credentials/guidance/:type - Get guidance for credential type
      - GET /api/credentials/guidance - List all credential guidance templates
      - POST /api/credentials/guidance - Create guidance template (admin)
      - PUT /api/credentials/guidance/:id - Update guidance template (admin)
      - DELETE /api/credentials/guidance/:id - Delete guidance template (admin)
    </credential_guidance>

    <admin>
      - GET /api/admin/users - List all users
      - PUT /api/admin/users/:id/activate - Activate user
      - PUT /api/admin/users/:id/deactivate - Deactivate user
      - GET /api/admin/audit-log - Get company-wide audit log with filters
      - GET /api/admin/statistics - Get system statistics
      - GET /api/admin/export/audit-log - Export audit log as CSV
    </admin>

    <websocket_events>
      - workflow:progress - Real-time workflow generation progress updates
      - workflow:complete - Workflow generation completed
      - workflow:error - Workflow generation failed
    </websocket_events>
  </api_endpoints_summary>

  <mcp_integration>
    <architecture>
      - Backend runs n8n-mcp server as a child process or service
      - MCP server configured with environment variables (N8N_API_URL, N8N_API_KEY per request)
      - Gemini AI communicates with MCP server using Model Context Protocol
      - MCP provides tools for: listing nodes, getting node details, creating workflows
      - Backend manages MCP server lifecycle and error handling
    </architecture>

    <mcp_tools_used>
      - n8n_list_nodes: Discover available nodes in target instance
      - n8n_get_node: Get detailed node information
      - n8n_create_workflow: Create workflow in n8n instance
      - n8n_activate_workflow: Activate created workflow
    </mcp_tools_used>

    <session_management>
      - Each workflow generation gets a unique session ID
      - MCP server instance reused across requests (pooling)
      - API keys passed dynamically per request (not hardcoded)
      - Timeout handling for long-running AI operations
    </session_management>
  </mcp_integration>
